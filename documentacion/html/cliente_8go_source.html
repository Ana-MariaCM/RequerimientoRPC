22/**
23 * @brief Crea la conexión gRPC, construye las capas del cliente y muestra el menú principal.
24 */
25func main() {
26 conexion, err := grpc.NewClient(configuracion.ObtenerDireccionStreaming(),
27 grpc.WithTransportCredentials(insecure.NewCredentials()))
28 if err != nil {
29 fmt.Println("No fue posible crear la conexión con el servidor de streaming:", err)
30 os.Exit(1)
31 }
32 defer conexion.Close()
33
34 fachadaMetadatos := fachada.NuevaFachadaMetadatos(configuracion.ObtenerURLMetadatos())
35 fachadaStreaming := fachada.NuevaFachadaStreaming(pb.NewAudioServiceClient(conexion), obtenerUsuario())
36
37 controladorAudios := capacontroladores.NuevoControladorAudios(fachadaMetadatos)
38 controladorReproduccion := capacontroladores.NuevoControladorReproduccion(fachadaStreaming)
39
40 vistas.NuevoNavegadorVistas(controladorAudios, controladorReproduccion).MostrarMenuPrincipal()
41}

44 * @brief Identifica al usuario como usuario\@equipo para las estadísticas.

47func obtenerUsuario() string {
48 nombreUsuario := "anonimo"
49 if usuarioActual, err := user.Current(); err == nil {
50 nombreUsuario = usuarioActual.Username
51 }
52 equipo, err := os.Hostname()
53 if err != nil {
54 return nombreUsuario
55 }
56 return nombreUsuario + "@" + equipo
57}
