15/** @brief Lector compartido de la entrada estándar. */
16var lectorEntrada = bufio.NewReader(os.Stdin)

18/**
19 * @brief Muestra un mensaje y lee una línea de texto.
20 * @param mensaje Texto que se muestra antes de leer.
21 * @return Texto ingresado sin espacios al inicio ni al final.
22 */
23func LeerTexto(mensaje string) string {
24 fmt.Print(mensaje)
25 linea, err := lectorEntrada.ReadString('\n')
26 if err != nil && linea == "" {
27 // Fin de la entrada estándar: se termina la aplicación.
28 fmt.Println()
29 os.Exit(0)
30 }
31 return strings.TrimSpace(linea)
32}

34/**
35 * @brief Lee una opción numérica comprendida entre un mínimo y un máximo.
36 *
37 * Repite la lectura hasta que el usuario ingrese un número válido.
38 * @param mensaje Texto que se muestra antes de leer.
39 * @param minimo Valor mínimo permitido.
40 * @param maximo Valor máximo permitido.
41 * @return Opción seleccionada.
42 */
43func LeerOpcion(mensaje string, minimo int, maximo int) int {
44 for {
45 opcion, err := strconv.Atoi(LeerTexto(mensaje))
46 if err == nil && opcion >= minimo && opcion <= maximo {
47 return opcion
48 }
49 fmt.Printf("Opción no válida, ingrese un número entre %d y %d.\n", minimo, maximo)
50 }
51}

53/**
54 * @brief Convierte una cantidad de bytes en un texto legible (B, KB o MB).
55 * @param bytes Cantidad de bytes.
56 * @return Texto con la unidad adecuada.
57 */
58func FormatearTamanio(bytes int64) string {
59 switch {
60 case bytes >= 1<<20:
61 return fmt.Sprintf("%.2f MB", float64(bytes)/(1<<20))
62 case bytes >= 1<<10:
63 return fmt.Sprintf("%.1f KB", float64(bytes)/(1<<10))
64 default:
65 return fmt.Sprintf("%d B", bytes)
66 }
