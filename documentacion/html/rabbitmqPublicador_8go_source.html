18/**
19 * @brief Publicador de mensajes JSON en una cola de RabbitMQ.
20 */
21type RabbitPublicador struct {
22 mu sync.Mutex ///< Serializa las publicaciones (el canal AMQP no es concurrente).
23 url string ///< URL de conexión al broker.
24 nombreCola string ///< Nombre de la cola destino.
25 conexion *amqp.Connection ///< Conexión con el broker.
26 canal *amqp.Channel ///< Canal AMQP abierto sobre la conexión.
27}

29/**
30 * @brief Crea un publicador y establece la conexión con RabbitMQ.
31 * @param url URL AMQP del broker (por ejemplo amqp://guest:guest\@localhost:5672/).
32 * @param nombreCola Nombre de la cola en la que se publicarán los mensajes.
33 * @return El publicador creado. Si la conexión falla se devuelve también el error;
34 * el publicador intentará reconectarse en la siguiente publicación.
35 */
36func NuevoRabbitPublicador(url string, nombreCola string) (*RabbitPublicador, error) {
37 publicador := &RabbitPublicador{url: url, nombreCola: nombreCola}
38 publicador.mu.Lock()
39 defer publicador.mu.Unlock()
40 return publicador, publicador.conectar()
41}

43/**
44 * @brief Abre la conexión, el canal y declara la cola (durable).
45 * @return Error si no fue posible conectarse con el broker.
46 */
47func (thisP *RabbitPublicador) conectar() error {
48 conexion, err := amqp.Dial(thisP.url)
49 if err != nil {
50 return fmt.Errorf("error conectando a RabbitMQ: %v", err)
51 }
52
53 canal, err := conexion.Channel()
54 if err != nil {
55 conexion.Close()
56 return fmt.Errorf("error abriendo canal: %v", err)
57 }
58
59 _, err = canal.QueueDeclare(
60 thisP.nombreCola, // nombre
61 true, // durable
62 false, // autoDelete
63 false, // exclusive
64 false, // noWait
65 nil, // argumentos
66 )
67 if err != nil {
68 canal.Close()
69 conexion.Close()
70 return fmt.Errorf("error declarando cola: %v", err)
71 }
72
73 thisP.conexion = conexion
74 thisP.canal = canal
75 return nil
76}

78/**
79 * @brief Serializa un mensaje a JSON y lo publica en la cola.
80 *
81 * Si no existe una conexión activa se intenta reconectar antes de publicar.
82 * @param mensaje Objeto (DTO) a publicar.
83 * @return Error si no se pudo serializar o publicar el mensaje.
84 */
85func (thisP *RabbitPublicador) Publicar(mensaje any) error {
86 cuerpo, err := json.Marshal(mensaje)
87 if err != nil {
88 return fmt.Errorf("error convirtiendo mensaje a JSON: %v", err)
89 }
90
91 thisP.mu.Lock()
92 defer thisP.mu.Unlock()
93
94 if thisP.conexion == nil || thisP.conexion.IsClosed() {
95 if err := thisP.conectar(); err != nil {
96 return err
97 }
98 }
99
100 err = thisP.canal.Publish(
101 "", // exchange por defecto
102 thisP.nombreCola, // routing key = nombre de la cola
103 false, // mandatory
104 false, // immediate
105 amqp.Publishing{
106 ContentType: "application/json",
107 DeliveryMode: amqp.Persistent,
108 Body: cuerpo,
109 },
110 )
111 if err != nil {
112 thisP.conexion = nil
113 return fmt.Errorf("error publicando mensaje: %v", err)
114 }
115
116 fmt.Printf("[Cola] Mensaje publicado en la cola \"%s\": %s\n", thisP.nombreCola, string(cuerpo))
117 return nil
118}
