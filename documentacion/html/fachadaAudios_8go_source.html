38/**
39 * @brief Envía un archivo mp3 al servidor de audios (POST /audios/almacenamiento).
40 * @param audio Datos del audio y ruta local del archivo.
41 * @return Respuesta del servidor o un error.
42 */
43func (thisF *FachadaAudios) AlmacenarAudio(audio dtos.AudioAlmacenarDTO) (dtos.AudioAlmacenadoDTO, error) {
44 var respuesta dtos.AudioAlmacenadoDTO
45
46 archivo, err := os.Open(audio.RutaLocal)
47 if err != nil {
48 return respuesta, fmt.Errorf("no se pudo abrir el archivo local: %v", err)
49 }
50 defer archivo.Close()
51
52 cuerpo := &bytes.Buffer{}
53 formulario := multipart.NewWriter(cuerpo)
54 campoArchivo, err := formulario.CreateFormFile("archivo", filepath.Base(audio.RutaLocal))
55 if err != nil {
56 return respuesta, err
57 }
58 if _, err := io.Copy(campoArchivo, archivo); err != nil {
59 return respuesta, fmt.Errorf("no se pudo leer el archivo local: %v", err)
60 }
61 formulario.WriteField("titulo", audio.Titulo)
62 formulario.WriteField("tipo", audio.Tipo)
63 formulario.WriteField("nombreArchivo", audio.NombreArchivo)
64 formulario.Close()
65
66 peticion, err := http.NewRequest(http.MethodPost, thisF.urlBase+"/audios/almacenamiento", cuerpo)
67 if err != nil {
68 return respuesta, err
69 }
70 peticion.Header.Set("Content-Type", formulario.FormDataContentType())
71
72 err = thisF.ejecutar(peticion, http.StatusCreated, &respuesta)
73 return respuesta, err
74}

91/**
92 * @brief Ejecuta una petición HTTP y deserializa la respuesta JSON.
93 * @param peticion Petición a ejecutar.
94 * @param codigoEsperado Código HTTP que indica éxito.
95 * @param destino Puntero donde se deserializa la respuesta.
96 * @return Error de conexión, el mensaje de error del servidor, o nil.
97 */
98func (thisF *FachadaAudios) ejecutar(peticion *http.Request, codigoEsperado int, destino any) error {
99 respuestaHTTP, err := thisF.clienteWeb.Do(peticion)
100 if err != nil {
101 return fmt.Errorf("no fue posible comunicarse con el servidor de audios (%s)", thisF.urlBase)
102 }
103 defer respuestaHTTP.Body.Close()
104
105 if respuestaHTTP.StatusCode != codigoEsperado {
106 var errorDTO dtos.RespuestaErrorDTO
107 if json.NewDecoder(respuestaHTTP.Body).Decode(&errorDTO) == nil && errorDTO.Mensaje != "" {
108 return fmt.Errorf("%s", errorDTO.Mensaje)
109 }
110 return fmt.Errorf("el servidor respondió con el código %d", respuestaHTTP.StatusCode)
111 }
112
113 if err := json.NewDecoder(respuestaHTTP.Body).Decode(destino); err != nil {
114 return fmt.Errorf("respuesta inválida del servidor: %v", err)
115 }
116 return nil
117}
