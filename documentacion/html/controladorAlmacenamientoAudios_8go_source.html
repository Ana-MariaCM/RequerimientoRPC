34/**
35 * @brief Servicio REST POST /audios/almacenamiento: almacena un nuevo audio mp3.
36 *
37 * Recibe un formulario multipart con los campos "archivo" (mp3), "titulo",
38 * "tipo" y "nombreArchivo".
39 * @param w Escritor de la respuesta HTTP.
40 * @param r Petición HTTP recibida.
41 */
42func (thisC *ControladorAlmacenamientoAudios) AlmacenarAudio(w http.ResponseWriter, r *http.Request) {
43 fmt.Printf("\n[REST] POST /audios/almacenamiento invocado por %s\n", r.RemoteAddr)
44
45 r.Body = http.MaxBytesReader(w, r.Body, configuracion.TamanioMaximoAudio)
46 if err := r.ParseMultipartForm(32 << 20); err != nil {
47 fmt.Println("[REST] Formulario inválido:", err)
48 responderError(w, http.StatusBadRequest, "El formulario es inválido o supera el tamaño máximo permitido")
49 return
50 }
51
52 archivo, _, err := r.FormFile("archivo")
53 if err != nil {
54 fmt.Println("[REST] No se recibió el campo \"archivo\"")
55 responderError(w, http.StatusBadRequest, "Error leyendo el archivo")
56 return
57 }
58 defer archivo.Close()
59
60 datos, err := io.ReadAll(archivo)
61 if err != nil {
62 responderError(w, http.StatusBadRequest, "Error leyendo el archivo")
63 return
64 }
65
66 audioDTO := dtos.AudioAlmacenarDTOInput{
67 Titulo: r.FormValue("titulo"),
68 Tipo: r.FormValue("tipo"),
69 NombreArchivo: r.FormValue("nombreArchivo"),
70 }
71 fmt.Printf("[REST] Almacenando audio \"%s\" [%s] (%d bytes)\n", audioDTO.Titulo, audioDTO.Tipo, len(datos))
72
73 respuesta, err := thisC.fachada.GuardarAudio(audioDTO, datos)
74 if errors.Is(err, capafachada.ErrAudioInvalido) {
75 fmt.Println("[REST]", err)
76 responderError(w, http.StatusBadRequest, err.Error())
77 return
78 }
79 if err != nil {
80 fmt.Println("[REST]", err)
81 responderError(w, http.StatusInternalServerError, err.Error())
82 return
83 }
84
85 fmt.Printf("[REST] POST /audios/almacenamiento -> audio almacenado como %s\n", respuesta.Archivo.NombreArchivo)
86 responderJSON(w, http.StatusCreated, respuesta)
87}
