55/**
56 * @brief Transmite un archivo de audio al cliente fragmento a fragmento.
57 *
58 * Antes de iniciar el envío publica, de forma asíncrona, la información de la
59 * reproducción en la cola de mensajes. Si el cliente abandona la reproducción
60 * el envío se detiene.
61 * @param peticion Petición gRPC con el archivo y los datos del audio.
62 * @param direccionCliente Dirección IP:puerto del cliente.
63 * @param stream Flujo gRPC por el que se envían los fragmentos.
64 * @return nil si el audio se envió completo, o un error gRPC (NotFound, Canceled, Internal).
65 */
66func (thisF *FachadaStreaming) TransmitirAudio(peticion *pb.AudioRequest, direccionCliente string, stream pb.AudioService_AudioStreamServer) error {
67 archivo, err := thisF.repositorio.AbrirArchivo(peticion.Filename)
68 if err != nil {
69 fmt.Printf("[gRPC] No se encontró el audio \"%s\"\n", peticion.Filename)
70 return status.Errorf(codes.NotFound, "el audio %s no está disponible en el servidor", peticion.Filename)
71 }
72 defer archivo.Close()
73
74 // Comunicación asíncrona: la publicación no bloquea el streaming.
75 go thisF.publicarReproduccion(peticion, direccionCliente)
76
77 return thisF.enviarFragmentos(archivo, stream)
78}

80/**
81 * @brief Lee el archivo en fragmentos de 32 KB y los envía por el stream gRPC.
82 * @param lector Archivo de audio abierto.
83 * @param stream Flujo gRPC por el que se envían los fragmentos.
84 * @return nil si se envió completo, o un error gRPC.
85 */
86func (thisF *FachadaStreaming) enviarFragmentos(lector io.Reader, stream pb.AudioService_AudioStreamServer) error {
87 buffer := make([]byte, configuracion.TamanioFragmento)
88 numeroFragmento := int32(0)
89
90 for {
91 n, err := lector.Read(buffer)
92 if n > 0 {
93 numeroFragmento++
94 fragmento := &pb.AudioChunk{Data: buffer[:n], NumeroFragmento: numeroFragmento}
95 if errEnvio := stream.Send(fragmento); errEnvio != nil {
96 if stream.Context().Err() != nil {
97 fmt.Printf("[gRPC] El cliente abandonó la reproducción tras %d fragmentos\n", numeroFragmento-1)
98 return status.Error(codes.Canceled, "reproducción cancelada por el cliente")
99 }
100 return status.Errorf(codes.Internal, "error enviando el fragmento #%d: %v", numeroFragmento, errEnvio)
101 }
102 fmt.Printf("[gRPC] Fragmento #%d enviado (%d bytes)\n", numeroFragmento, n)
103 }
104 if err == io.EOF {
105 fmt.Printf("[gRPC] Audio enviado completo (%d fragmentos)\n", numeroFragmento)
106 return nil
107 }
108 if err != nil {
109 return status.Errorf(codes.Internal, "error leyendo el archivo: %v", err)
110 }
111 }
112}
