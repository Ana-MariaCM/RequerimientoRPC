44/**
45 * @brief Opción 1: solicita los datos de un audio y lo envía al servidor de audios.
46 * @param controlador Controlador del administrador.
47 */
48func opcionAlmacenarAudio(controlador *capacontroladores.ControladorAdministrador) {
49 fmt.Println("\n--- Almacenar audio ---")
50 audio := dtos.AudioAlmacenarDTO{}
51 rutaLocal, err := controlador.ValidarArchivoLocal(utilidades.LeerTexto("Ruta del archivo mp3: "))
52 if err != nil {
53 fmt.Println("No se puede almacenar el audio:", err)
54 return
55 }
56 audio.RutaLocal = rutaLocal
57
58 fmt.Println("Tipo de audio:")
59 for indice, tipo := range tiposAudio {
60 fmt.Printf(" %d. %s\n", indice+1, tipo)
61 }
62 audio.Tipo = tiposAudio[utilidades.LeerOpcion("Seleccione el tipo: ", 1, len(tiposAudio))-1]
63 audio.Titulo = utilidades.LeerTexto("Título del audio: ")
64 audio.NombreArchivo = utilidades.LeerTexto(fmt.Sprintf("Nombre en el servidor (Enter = %s): ", filepath.Base(audio.RutaLocal)))
65
66 respuesta, err := controlador.AlmacenarAudio(audio)
67 if err != nil {
68 fmt.Println("No se pudo almacenar el audio:", err)
69 return
70 }
71 fmt.Printf("%s: %s (%s)\n", respuesta.Mensaje, respuesta.Archivo.NombreArchivo,
72 utilidades.FormatearTamanio(respuesta.Archivo.TamanioBytes))
73}
