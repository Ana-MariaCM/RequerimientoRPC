28/**
29 * @brief Reproductor de audio mp3 alimentado por streaming.
30 */
31type Reproductor struct {
32 lector *io.PipeReader ///< Extremo de lectura de la tubería (lo consume el decodificador).
33 escritor *io.PipeWriter ///< Extremo de escritura de la tubería (lo alimentan los fragmentos).
34 finalizado chan struct{} ///< Se cierra cuando la reproducción termina o se detiene.
35 unaVez sync.Once ///< Garantiza que el canal finalizado se cierre una sola vez.
36 mu sync.Mutex ///< Protege el estado detenido durante la inicialización del altavoz.
37 detenido bool ///< Indica si el usuario abandonó la reproducción.
38 muError sync.Mutex ///< Protege el campo err.
39 err error ///< Primer error ocurrido durante la recepción o la reproducción.
40}

51/**
52 * @brief Decodifica el mp3 a medida que llegan los fragmentos y lo envía al altavoz.
53 *
54 * Se debe ejecutar en una goroutine: bloquea hasta recibir la cabecera del mp3.
55 */
56func (thisR *Reproductor) Reproducir() {
57 streamer, formato, err := mp3.Decode(thisR.lector)
58 if err != nil {
59 thisR.finalizarConError(fmt.Errorf("no se pudo decodificar el audio: %v", err))
60 return
61 }
62
63 thisR.mu.Lock()
64 if thisR.detenido {
65 thisR.mu.Unlock()
66 streamer.Close()
67 return
68 }
69
70 errAltavoz := speaker.Init(formato.SampleRate, formato.SampleRate.N(time.Second/10))
71 if errAltavoz == nil {
72 speaker.Play(beep.Seq(streamer, beep.Callback(func() {
73 if errStreamer := streamer.Err(); errStreamer != nil && !errors.Is(errStreamer, errReproduccionDetenida) {
74 thisR.registrarError(errStreamer)
75 }
76 thisR.marcarFinalizado()
77 })))
78 }
79 thisR.mu.Unlock()
80
81 if errAltavoz != nil {
82 streamer.Close()
83 thisR.finalizarConError(fmt.Errorf("no se pudo inicializar el dispositivo de audio: %v", errAltavoz))
84 }
85}

87/**
88 * @brief Detiene la reproducción en cualquier momento.
89 *
90 * Primero cierra la tubería para desbloquear al decodificador y luego retira
91 * el audio del altavoz.
92 */
93func (thisR *Reproductor) Detener() {
94 thisR.mu.Lock()
95 thisR.detenido = true
96 thisR.mu.Unlock()
97
98 thisR.lector.CloseWithError(errReproduccionDetenida)
99 speaker.Clear()
100 thisR.marcarFinalizado()
101}

133/**
134 * @brief Registra un error, cierra la tubería y marca la reproducción como finalizada.
135 * @param err Error ocurrido.
136 */
137func (thisR *Reproductor) finalizarConError(err error) {
138 thisR.mu.Lock()
139 detenido := thisR.detenido
140 thisR.mu.Unlock()
141 if !detenido {
142 thisR.registrarError(err)
143 }
144 thisR.lector.CloseWithError(err)
145 thisR.marcarFinalizado()
146}

151func (thisR *Reproductor) marcarFinalizado() {
152 thisR.unaVez.Do(func() { close(thisR.finalizado) })

155/**
156 * @brief Recibe los fragmentos del servidor de streaming y los entrega al reproductor.
157 *
158 * Termina cuando se recibe el audio completo, cuando ocurre un error o cuando
159 * el usuario abandona la reproducción.
160 * @param stream Flujo gRPC del procedimiento remoto AudioStream.
161 * @param primerFragmento Fragmento recibido al invocar el procedimiento remoto.
162 * @param reproductor Reproductor que consume los fragmentos.
163 */
164func RecibirAudio(stream pb.AudioService_AudioStreamClient, primerFragmento *pb.AudioChunk, reproductor *Reproductor) {
165 fragmento := primerFragmento
166 for {
167 if _, err := reproductor.escritor.Write(fragmento.Data); err != nil {
168 // El reproductor cerró la tubería (usuario salió o error de audio).
169 return
170 }
171
172 siguiente, err := stream.Recv()
173 if err == io.EOF {
174 reproductor.escritor.Close()
175 return
176 }
177 if err != nil {
178 if stream.Context().Err() == nil {
179 reproductor.registrarError(fmt.Errorf("se interrumpió la recepción del audio"))
180 }
181 reproductor.escritor.CloseWithError(err)
182 return
183 }
184 fragmento = siguiente
185 }
