79/**
80 * @brief Convierte un mapa de conteos en una lista ordenada de mayor a menor.
81 * @param conteos Mapa nombre -> cantidad.
82 * @return Lista de conteos ordenada por cantidad (y por nombre en caso de empate).
83 */
84func ordenarConteos(conteos map[string]int) []modelos.Conteo {
85 lista := make([]modelos.Conteo, 0, len(conteos))
86 for nombre, cantidad := range conteos {
87 lista = append(lista, modelos.Conteo{Nombre: nombre, Cantidad: cantidad})
88 }
89 sort.Slice(lista, func(i, j int) bool {
90 if lista[i].Cantidad != lista[j].Cantidad {
91 return lista[i].Cantidad > lista[j].Cantidad
92 }
93 return lista[i].Nombre < lista[j].Nombre
94 })
95 return lista
