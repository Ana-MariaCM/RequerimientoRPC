65/**

71func (thisF *FachadaMetadatos) consultar(ruta string, destino any) error {
72 respuesta, err := thisF.clienteWeb.Get(thisF.urlBase + ruta)
73 if err != nil {
74 return fmt.Errorf("no fue posible comunicarse con el servidor de metadatos (%s)", thisF.urlBase)
75 }
76 defer respuesta.Body.Close()
77
78 if respuesta.StatusCode != http.StatusOK {
79 var errorDTO dtos.RespuestaErrorDTO
80 if json.NewDecoder(respuesta.Body).Decode(&errorDTO) == nil && errorDTO.Mensaje != "" {
81 return fmt.Errorf("%s", errorDTO.Mensaje)
82 }
83 return fmt.Errorf("el servidor de metadatos respondió con el código %d", respuesta.StatusCode)
84 }
85
86 if err := json.NewDecoder(respuesta.Body).Decode(destino); err != nil {
87 return fmt.Errorf("respuesta inválida del servidor de metadatos: %v", err)
88 }
89 return nil
