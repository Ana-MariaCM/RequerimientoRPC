27/**
28 * @brief Conecta con RabbitMQ y declara la cola (durable) de la que se consumirá.
29 * @param url URL AMQP del broker.
30 * @param nombreCola Nombre de la cola.
31 * @return El consumidor conectado o un error si no fue posible conectarse.
32 */
33func NuevoRabbitConsumidor(url string, nombreCola string) (*RabbitConsumidor, error) {
34 conexion, err := amqp.Dial(url)
35 if err != nil {
36 return nil, fmt.Errorf("error conectando a RabbitMQ: %v", err)
37 }
38
39 canal, err := conexion.Channel()
40 if err != nil {
41 conexion.Close()
42 return nil, fmt.Errorf("error abriendo canal: %v", err)
43 }
44
45 _, err = canal.QueueDeclare(nombreCola, true, false, false, false, nil)
46 if err != nil {
47 canal.Close()
48 conexion.Close()
49 return nil, fmt.Errorf("error declarando cola: %v", err)
50 }
51
52 // Se entrega un mensaje a la vez para no sobrecargar el servidor.
53 if err := canal.Qos(1, 0, false); err != nil {
54 canal.Close()
55 conexion.Close()
56 return nil, fmt.Errorf("error configurando QoS: %v", err)
57 }
58
59 return &RabbitConsumidor{conexion: conexion, canal: canal, nombreCola: nombreCola}, nil
60}

62/**
63 * @brief Consume mensajes de la cola indefinidamente.
64 *
65 * Por cada mensaje imprime un eco, invoca el manejador y confirma (ack) el
66 * mensaje. La función retorna cuando se cierra la conexión con el broker.
67 * @param manejador Función que procesa el cuerpo de cada mensaje.
68 * @return Error si no se pudo registrar el consumidor.
69 */
70func (thisC *RabbitConsumidor) Consumir(manejador ManejadorMensaje) error {
71 mensajes, err := thisC.canal.Consume(
72 thisC.nombreCola, // cola
73 "", // consumidor
74 false, // autoAck (se confirma manualmente)
75 false, // exclusive
76 false, // noLocal
77 false, // noWait
78 nil, // argumentos
79 )
80 if err != nil {
81 return fmt.Errorf("error registrando el consumidor: %v", err)
82 }
83
84 for mensaje := range mensajes {
85 fmt.Printf("\n[Cola] Mensaje consumido de la cola \"%s\": %s\n", thisC.nombreCola, string(mensaje.Body))
86
87 if err := manejador(mensaje.Body); err != nil {
88 fmt.Println("[Cola] Mensaje descartado:", err)
89 mensaje.Nack(false, false)
90 continue
91 }
92 mensaje.Ack(false)
93 }
94 return nil
95}
